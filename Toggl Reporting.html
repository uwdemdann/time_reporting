def fn_tbl_by_mission(df_time_entries):
    st_dt = min(df_time_entries.select(pl.col('start'))[:,'start'].str.to_datetime(time_zone="UTC").dt.date())
    en_dt = max(df_time_entries.select(pl.col('start'))[:,'start'].str.to_datetime(time_zone="UTC").dt.date())
    d = df_time_entries.group_by(
        ['toggl_user_name', 'client_name']
    ).agg(
        (pl.sum('duration') / 3600).alias('hours')
    ).filter(
        ~pl.col('client_name').is_null(),
        ~pl.col('client_name').eq('PTO')
    ).select(
        pl.col('toggl_user_name').alias('Analyst'),
        pl.col('client_name').alias('Mission'),
        pl.col('hours'),
        (pl.col('hours') / pl.sum('hours').over('toggl_user_name')).alias('percent')
    ).pivot(
        on='Analyst',
        index='Mission',
        values=[ 'hours', 'percent' ]
    ).sort('Mission').select(
        pl.col('Mission'),
        cs.ends_with('Adam'),
        cs.ends_with('Dann')
    )
    #
    t = gt(d).cols_label(
        Mission="Mission",
        hours_Dann="Hours",
        hours_Adam="Hours",
        percent_Adam="Percent of Total",
        percent_Dann="Percent of Total"
    ).fmt_percent(
        columns=cs.starts_with('percent'),
        decimals=1
    ).fmt_number(
        columns=cs.starts_with('hour'),
        decimals=1
    ).tab_spanner(
        columns=cs.ends_with('Adam'),
        label="Adam"
    ).tab_spanner(
        columns=cs.ends_with('Dann'),
        label="Dann"
    ).sub_missing(
        columns = cs.all(),
        missing_text = '---'
    ).tab_footnote(
        f"Only includes entries from {st_dt.strftime('%x')} to {en_dt.strftime('%x')}.\nExcludes PTO time."
    )
    return t
